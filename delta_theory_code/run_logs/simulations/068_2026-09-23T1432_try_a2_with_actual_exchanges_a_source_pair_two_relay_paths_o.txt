import numpy as np
def run(L1,L2,lat,T=6000,amp=0.02,order='outward'):
    # nodes: 0 = S1, 1 = S2 (source pair), path A: S1 - a_1..a_L1 - R, path B: S1 - b_1..b_L2 - R
    edges=[(0,1)]; n=2
    a=[0]+list(range(n,n+L1)); n+=L1
    b=[0]+list(range(n,n+L2)); n+=L2
    R=n; n+=1
    for p in (a,b):
        for u,v in zip(p,p[1:]): edges.append((u,v))
        edges.append((p[-1],R))
    tau=np.ones(n); dl=np.ones(n)
    tau[0]*=1+amp; dl[0]*=1-amp; tau[1]*=1-amp; dl[1]*=1+amp     # the source pair starts slightly out of balance
    hist=[dl.copy() for _ in range(lat+1)]

    return ok,np.array(thR)
