import numpy as np
from perspectival import colour
# A ring of 10 identical members (a nearly even bound structure), one halving's worth of difference at member 0.
M=10; edges=[(k,(k+1)%M) for k in range(M)]; col,C=colour(edges)
def run(src,R=40000,seed=0,noise=1e-3):
    rng=np.random.default_rng(seed)
    tau=np.exp(noise*rng.normal(size=M)); dl=np.exp(noise*rng.normal(size=M))
    tau[src]*=np.sqrt(2); dl[src]/=np.sqrt(2)
    rel={(u,v):(tau[u]/(tau[u]+tau[v]),dl[u]/(dl[u]+dl[v])) for (u,v) in edges}
    T0,D0=tau.sum(),dl.sum()
    exc=np.zeros(M); th2=np.zeros(M); q=np.zeros(M)
    for r in range(R):
        for t in range(C):
            for (u,v) in edges:
                if col[(u,v)]!=t: continue
                T=tau[u]+tau[v]; D=dl[u]+dl[v]; xu,yu=rel[(u,v)]; rel[(u,v)]=(tau[u]/T,dl[u]/D)
                tau[u],tau[v]=T*yu,T*(1-yu); dl[u],dl[v]=D*xu,D*(1-xu)
        lt=np.log(tau/(T0/M)); ld=np.log(dl/(D0/M))
        th=0.5*(lt-ld); th-=th.mean()                   # rapidity in the structure's own frame
        exc+=np.exp(-th)-1                              # extra registration by a detector at rest with the structure
        th2+=th**2
        dens=lt**2+ld**2                                # the conserved total's density: parts ...
        for (u,v),(x,y) in rel.items():                 # ... plus the relations' memory, shared between the two ends
            w=8*((x-0.5)**2+(y-0.5)**2); dens[u]+=w/2; dens[v]+=w/2
        q+=dens
    return exc/R,th2/R,q/R
for src in (0,3):
    exc,th2,q=run(src)
    pe=exc/exc.sum(); pt=th2/th2.sum(); pq=q/q.sum()

    outcomes=rng.choice(M,size=N,p=pe)             # Postulate 8: the whole quantum is registered at one place, with the
    f=np.bincount(outcomes,minlength=M)/N           # probability of each place given by its extra registration
    chi=N*np.sum((f-pq)**2/pq)
    print(f"quantum placed at member {src}:")
    print("  outcome frequencies (5,000 samples):",np.round(f,3).tolist())

    print(f"  extra registration vs squared rapidity: max difference in shares {np.abs(pe-pt).max():.4f}; "
          f"vs conserved total: {np.abs(pe-pq).max():.4f}; chi-square of outcomes against the conserved total {chi:.1f} on 9 d.f.")
