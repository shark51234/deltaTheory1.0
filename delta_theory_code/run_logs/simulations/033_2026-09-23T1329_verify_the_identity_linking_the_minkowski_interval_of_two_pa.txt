# Check the identity: (tau_i - tau_j)(delta_i - delta_j) = (M_ii + M_jj) - (M_ij + M_ji)
import numpy as np
r=np.random.default_rng(0); worst=0
for _ in range(100000):
    ti,di,tj,dj=r.uniform(0.01,2,4)
    lhs=(ti-tj)*(di-dj); rhs=(ti*di+tj*dj)-(ti*dj+tj*di)
    worst=max(worst,abs(lhs-rhs))
print("max |interval - (selves - relation)| over 100,000 random pairs:",worst)
