import numpy as np
from interference2 import build
def round_map(state,edges,col,C,n):
    tau,dl,rec=state
    tau=tau.copy(); dl=dl.copy(); rec=dict(rec)
    for t in range(C):
        for (u,v) in edges:
            if col[(u,v)]!=t: continue
            du,dv=dl[u],dl[v]; F=tau[u]*rec[(u,v)]-tau[v]*rec[(v,u)]
            Tt=tau[u]+tau[v]; Dd=du+dv; dl[u]+=F/Tt; dl[v]-=F/Tt; tau[u]-=F/Dd; tau[v]+=F/Dd
            rec[(u,v)]=dv; rec[(v,u)]=du
    return tau,dl,rec
def jacobian(edges,col,n,C,eps=1e-7):
    keys=[(u,v) for (u,v) in edges]+[(v,u) for (u,v) in edges]
    def vec(s): return np.concatenate([s[0],s[1],[s[2][k] for k in keys]])
    def unvec(x): return (x[:n].copy(),x[n:2*n].copy(),{k:x[2*n+i] for i,k in enumerate(keys)})
    x0=vec((np.ones(n),np.ones(n),{k:1.0 for k in keys}))
    f0=vec(round_map(unvec(x0),edges,col,C,n)); J=np.zeros((len(x0),len(x0)))
    for i in range(len(x0)):
        x=x0.copy(); x[i]+=eps; J[:,i]=(vec(round_map(unvec(x),edges,col,C,n))-f0)/eps
    return J
def report(name,edges,n):

    return lam
