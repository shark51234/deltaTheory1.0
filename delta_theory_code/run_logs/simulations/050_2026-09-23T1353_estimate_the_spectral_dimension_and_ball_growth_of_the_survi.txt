G=json.load(open("geo_pair_4.json"))
for g in G:
    P=np.array(g['P']); t=np.arange(len(P)); b=np.array(g['balls'])
    def ds(t1,t2):
        a=np.mean(P[t1-1:t1+2]); c=np.mean(P[t2-1:t2+2])
        return -2*np.log(c/a)/np.log(t2/t1) if a>0 and c>0 else float('nan')

          f"ball exponent (r->2r) r=1..4: {np.round(D,2).tolist()} | growth factor per step r=1..7: {np.round(fac,2).tolist()}")
